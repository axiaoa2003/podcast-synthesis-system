# 支持逐句情绪调节的播客生成系统设计

## 技术可行性分析

### 现有系统基础
- 当前系统已实现了基于角色的情绪配置
- 支持MiniMax API的情绪参数传递
- 文稿解析和音频生成逻辑清晰

### 技术可行性
1. **MiniMax API支持**：API已支持情绪、语速、语调等参数的调节
2. **文稿解析扩展性**：当前parse_dialogue函数可扩展支持情绪参数解析
3. **前端交互可行性**：可通过增强编辑器功能实现逐句情绪配置
4. **性能影响**：逐句配置不会显著增加系统负载，因为原本就是逐句生成音频

## 技术方案设计

### 方案1：基于情绪参数标记的方案

#### 核心设计
- **文稿格式扩展**：在现有格式基础上添加情绪参数标记
  ```
  [A](happy,1.2,0): 你知道吗？我今天中了彩票！
  [B](surprised,1.0,0): 真的吗？太令人惊讶了！
  ```
  格式说明：`[角色](情绪类型,语速,语调): 文本内容`

- **解析逻辑修改**：扩展parse_dialogue函数，支持解析情绪参数
  ```python
  match = re.match(r'^\[([AB])\]\(([^,]+),([^,]+),([^)]+)\)\s*[:：]\s*(.*)$', line)
  ```

- **数据结构扩展**：
  ```python
  dialogue.append((role, text, emotion, speed, pitch))
  ```

#### 实现步骤
1. 修改`parse_dialogue`函数，支持解析情绪参数标记
2. 更新`generate_audio_segments`函数，使用每句的个性化情绪参数
3. 增强前端编辑器，支持可视化添加情绪参数标记
4. 添加情绪参数验证和修复功能

#### 优点
- 文稿格式直观，便于手动编辑和版本控制
- 与现有系统兼容性好，无需大幅修改
- 支持离线编辑和传输

#### 缺点
- 手动编辑情绪参数较为繁琐
- 语法错误可能导致解析失败

# podcast_utils.py
def parse_dialogue(script):
    """
    解析双人对话文稿，支持逐句情绪参数
    
    参数:
    - script: 播客文稿文本或JSON字符串
    
    返回:
    - 对话列表，每个元素为 (角色, 文本, 情绪, 语速, 语调) 元组
    """
    # 尝试解析为JSON
    try:
        import json
        data = json.loads(script)
        dialogues = data.get('dialogues', [])
        result = []
        for d in dialogues:
            result.append((
                d.get('role', 'A'),
                d.get('text', ''),
                d.get('emotion', ''),
                d.get('speed', 1.0),
                d.get('pitch', 0)
            ))
        return result
    except json.JSONDecodeError:
        # 解析为文本文稿
        pass
    
    # 现有解析逻辑，扩展支持情绪参数
    dialogue = []
    lines = script.split('\n')
    
    for line in lines:
        line = line.strip()
        if not line:
            continue
        
        # 匹配带情绪参数的格式：[A](happy,1.2,0): 文本内容
        match = re.match(r'^\[([AB])\]\(([^,]+),([^,]+),([^)]+)\)\s*[:：]\s*(.*)$', line)
        if match:
            role = match.group(1)
            emotion = match.group(2).strip()
            speed = float(match.group(3).strip())
            pitch = int(match.group(4).strip())
            text = match.group(5).strip()
            if text:
                dialogue.append((role, text, emotion, speed, pitch))
        else:
            # 匹配不带情绪参数的格式：[A]: 文本内容
            match = re.match(r'^\[([AB])\]\s*[:：]\s*(.*)$', line)
            if match:
                role = match.group(1)
                text = match.group(2).strip()
                if text:
                    dialogue.append((role, text, '', 1.0, 0))
            else:
                # 合并到上一行
                if dialogue:
                    prev_role, prev_text, prev_emotion, prev_speed, prev_pitch = dialogue[-1]
                    dialogue[-1] = (prev_role, f"{prev_text} {line}", prev_emotion, prev_speed, prev_pitch)
    
    return dialogue
```

def generate_audio_segments(dialogue, voice_config, minimax_client, output_dir=SYSTEM_CONFIG["output_dir"]):
    """
    逐句合成音频片段，支持逐句情绪参数
    
    参数:
    - dialogue: 对话列表，每个元素为 (角色, 文本, 情绪, 语速, 语调) 元组
    - voice_config: 角色到声音配置的映射，包含voice_id
    - minimax_client: MiniMaxClient 实例
    - output_dir: 输出目录
    
    返回:
    - 生成的音频文件路径列表
    """
    audio_files = []
    os.makedirs(output_dir, exist_ok=True)
    
    for i, (role, text, emotion, speed, pitch) in enumerate(dialogue):
        if not text:
            continue
        
        # 获取当前角色的基础配置
        config = voice_config[role]
        voice_id = config.get('voice_id', 'male-qn-qingse')
        
        # 使用逐句的情绪参数，若为空则使用角色默认配置
        final_emotion = emotion or config.get('emotion', '')
        final_speed = speed or config.get('speed', 1.0)
        final_pitch = pitch or config.get('pitch', 0)
        
        # 调用 MiniMax API 合成音频
        result = minimax_client.synthesize_speech(
            text=text,
            voice_id=voice_id,
            emotion=final_emotion,
            speed=final_speed,
            pitch=final_pitch
        )
        
        # 保存音频文件...
    
    return audio_files
```

function addEmotionConfigListeners() {
    // 情绪选择事件
    document.querySelectorAll('[id^="emotion-"]').forEach(select => {
        select.addEventListener('change', updateScriptFromConfig);
    });
    
    // 语速滑块事件
    document.querySelectorAll('[id^="speed-"]').forEach(range => {
        range.addEventListener('input', function() {
            this.nextElementSibling.textContent = this.value;
            updateScriptFromConfig();
        });
    });
    
    // 语调滑块事件
    document.querySelectorAll('[id^="pitch-"]').forEach(range => {
        range.addEventListener('input', function() {
            this.nextElementSibling.textContent = this.value;
            updateScriptFromConfig();
        });
    });
}

function updateScriptFromConfig() {
    // 从配置面板获取所有参数
    const configs = [];
    document.querySelectorAll('[id^="emotion-"]').forEach((select, index) => {
        const emotion = select.value;
        const speed = document.getElementById(`speed-${index}`).value;
        const pitch = document.getElementById(`pitch-${index}`).value;
        configs.push({ emotion, speed, pitch });
    });
    
    // 解析当前文稿
    const script = document.getElementById('scriptEditor').value;
    const dialogues = parseScriptToDialogues(script);
    
    // 更新对话的情绪参数
    const updatedDialogues = dialogues.map((dialogue, index) => {
        return {
            ...dialogue,
            emotion: configs[index].emotion,
            speed: parseFloat(configs[index].speed),
            pitch: parseInt(configs[index].pitch)
        };
    });
    
    // 转换回文本文稿
    const updatedScript = dialoguesToScript(updatedDialogues);
    
    // 更新文本编辑器
    document.getElementById('scriptEditor').value = updatedScript;
}

### 3. 实现文稿格式转换

#### 文本文稿转结构化数据
```javascript
function parseScriptToDialogues(script) {
    const dialogues = [];
    const lines = script.split('\n');
    
    lines.forEach(line => {
        line = line.trim();
        if (!line) return;
        
        // 匹配带情绪参数的格式
        const match = line.match(/^\[([AB])\]\(([^,]+),([^,]+),([^)]+)\)\s*[:：]\s*(.*)$/);
        if (match) {
            dialogues.push({
                role: match[1],
                text: match[5].trim(),
                emotion: match[2].trim(),
                speed: parseFloat(match[3].trim()),
                pitch: parseInt(match[4].trim())
            });
        } else {
            // 匹配不带情绪参数的格式
            const simpleMatch = line.match(/^\[([AB])\]\s*[:：]\s*(.*)$/);
            if (simpleMatch) {
                dialogues.push({
                    role: simpleMatch[1],
                    text: simpleMatch[2].trim(),
                    emotion: '',
                    speed: 1.0,
                    pitch: 0
                });
            }
        }
    });
    
    return dialogues;
}
```

#### 结构化数据转文本文稿
```javascript
function dialoguesToScript(dialogues) {
    let script = '';
    dialogues.forEach(dialogue => {
        if (dialogue.emotion || dialogue.speed !== 1.0 || dialogue.pitch !== 0) {
            // 带情绪参数的格式
            script += `[${dialogue.role}](${dialogue.emotion || '',dialogue.speed,dialogue.pitch}): ${dialogue.text}\n`;
        } else {
            // 不带情绪参数的格式
            script += `[${dialogue.role}]: ${dialogue.text}\n`;
        }
    });
    return script;
}
```

```python
# podcast_utils.py
def validate_script(script):
    """
    验证文稿格式，支持逐句情绪参数
    
    参数:
    - script: 播客文稿文本
    
    返回:
    - 包含验证结果的字典
    """
    dialogues = parse_dialogue(script)
    
    if not dialogues:
        return {
            "valid": False,
            "message": "文稿中未找到有效的对话内容，请使用 [A] 和 [B] 标记角色"
        }
    
    # 检查角色分布
    role_counts = {"A": 0, "B": 0}
    for role, _, _, _, _ in dialogues:
        role_counts[role] += 1
    
    if role_counts["A"] == 0 or role_counts["B"] == 0:
        return {
            "valid": False,
            "message": "文稿中缺少一个或多个角色的对话"
        }
    
    # 检查情绪参数有效性
    valid_emotions = ["", "happy", "sad", "angry", "fearful", "disgusted", "surprised", "calm", "fluent", "whisper"]
    for i, (_, _, emotion, speed, pitch) in enumerate(dialogues):
        if emotion not in valid_emotions:
            return {
                "valid": False,
                "message": f"第 {i+1} 句情绪参数无效: {emotion}"
            }
        if speed < 0.5 or speed > 2.0:
            return {
                "valid": False,
                "message": f"第 {i+1} 句语速参数无效: {speed}，范围应在 0.5-2.0 之间"
            }
        if pitch < -12 or pitch > 12:
            return {
                "valid": False,
                "message": f"第 {i+1} 句语调参数无效: {pitch}，范围应在 -12-12 之间"
            }
    
    return {
        "valid": True,
        "message": f"文稿验证通过，包含 {len(dialogues)} 句对话，角色 A: {role_counts['A']} 句，角色 B: {role_counts['B']} 句",
        "dialogue_count": len(dialogues),
        "role_counts": role_counts
    }
```
